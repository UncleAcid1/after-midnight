cmake_minimum_required(VERSION 3.16)
# Build for macOS 11+ even on a newer Mac (otherwise the app refuses to open on older macOS).
set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum macOS version")
project(MidnightMansionHD C)

set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()

find_package(SDL2 REQUIRED)
find_package(OpenGL REQUIRED)

# No fused multiply-adds anywhere (Apple Silicon's compiler fuses by default): relay co-op
# computes the same frames on a Mac, a PC and a PS3, and they must get the same numbers
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-ffp-contract=off)
endif()

# SpriteWorld X fork (rendering/sprite engine)
file(GLOB SWX_SOURCES src/swx/*.c)
add_library(swx STATIC ${SWX_SOURCES})
target_include_directories(swx PUBLIC src/swx third_party/stb third_party/dr_libs)
target_link_libraries(swx PUBLIC SDL2::SDL2 OpenGL::GL)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(swx PRIVATE -Wall -Wno-unused-function)
endif()


# The game
#
# Game data (images, sounds, mansions) comes from the original app bundle; set
# MM_GAME_DATA_DIR to the folder that contains "Game Data" if it lives elsewhere.
set(MM_GAME_DATA_DIR "${CMAKE_SOURCE_DIR}/original/Midnight Mansion HD.app/Contents/Resources"
    CACHE PATH "Folder containing the original 'Game Data' folder")

file(GLOB GAME_SOURCES src/game/*.c)
if(APPLE)
  set(MM_ICON "${MM_GAME_DATA_DIR}/mm_icon.icns")
  add_executable(MidnightMansionHD MACOSX_BUNDLE ${GAME_SOURCES})
  set_target_properties(MidnightMansionHD PROPERTIES
    OUTPUT_NAME "Midnight Mansion HD"
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_SOURCE_DIR}/platform/macos/Info.plist.in")
  # Copy the game data and icon into the bundle's Resources after each build.
  add_custom_command(TARGET MidnightMansionHD POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${MM_GAME_DATA_DIR}/Game Data"
            "$<TARGET_BUNDLE_CONTENT_DIR:MidnightMansionHD>/Resources/Game Data"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/extras"
            "$<TARGET_BUNDLE_CONTENT_DIR:MidnightMansionHD>/Resources/Game Data Extras"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${MM_ICON}"
            "$<TARGET_BUNDLE_CONTENT_DIR:MidnightMansionHD>/Resources/mm_icon.icns"
    COMMAND "${CMAKE_SOURCE_DIR}/platform/macos/bundle_libs.sh" "$<TARGET_BUNDLE_DIR:MidnightMansionHD>"
    COMMENT "Copying Game Data and SDL2 into the app bundle")
  target_compile_definitions(MidnightMansionHD PRIVATE GL_SILENCE_DEPRECATION=1)
else()
  add_executable(MidnightMansionHD ${GAME_SOURCES})
endif()
target_include_directories(MidnightMansionHD PRIVATE src/game)
# Artwork and sounds this version adds (extras/, generated by tools/make_extras.py). The app
# bundle carries a copy as "Game Data Extras"; other builds read them from the source tree.
target_compile_definitions(MidnightMansionHD PRIVATE MM_EXTRAS_SOURCE_DIR="${CMAKE_SOURCE_DIR}/extras")
target_link_libraries(MidnightMansionHD PRIVATE swx)
if(NOT APPLE)
  target_link_libraries(MidnightMansionHD PRIVATE m)
endif()
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(MidnightMansionHD PRIVATE -Wall -Wno-unused-function -Wno-multichar
                         -Wno-missing-braces -Wno-unused-variable -Wno-pointer-sign
                         -Wno-format-truncation)
endif()
